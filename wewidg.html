    <script>

        // --- LOGIC ---

        const calendar = document.getElementById('calendar');

        const daysGrid = document.getElementById('days-grid');

        const monthDisplay = document.getElementById('month-display');

        const prevBtn = document.getElementById('prev');

        const nextBtn = document.getElementById('next');

        const detailMonth = document.getElementById('selected-month');

        const detailFull = document.getElementById('selected-date');

        

        let currentDate = new Date();

        

        // Month Names array for easy lookup

        const monthNames = ["January", "February", "March", "April", "May", "June", 

                           "July", "August", "September", "October", "November", "December"];


        function renderCalendar() {

            const currentYear = currentDate.getFullYear();

            const currentMonth = currentDate.getMonth();


            // Update Header

            monthDisplay.textContent = `${monthNames[currentMonth]} ${currentYear}`;

            

            // Determine days in current month

            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();

            

            // Determine padding from previous month

            const daysInPrevMonth = new Date(currentYear, currentMonth, 0).getDate();

            

            // Get day of week for the 1st of current month (0=Sun, 1=Mon)

            const firstDayIndex = new Date(currentYear, currentMonth, 1).getDay();


            let gridContainer = document.getElementById('days-grid');

            gridContainer.innerHTML = ''; // Clear previous days


            let today = new Date();


            // Render empty slots for previous month days

            for (let i = 0; i < firstDayIndex; i++) {

                const blank = document.createElement('div');

                blank.classList.add('day-cell', 'empty');

                gridContainer.appendChild(blank);

            }


            // Render current month days

            const todayDateObj = new Date();

            const todayDate = todayDateObj.getDate();


            // Create Date Objects for Today and Selection

            const todayDateObjFull = new Date(currentYear, currentMonth, todayDate);


            for (let i = 1; i <= daysInMonth; i++) {

                const dayDiv = document.createElement('div');

                dayDiv.classList.add('day-cell');

                dayDiv.textContent = i;

                

                // Check if it's today

                if (i === todayDate && currentMonth === today.getMonth() && currentYear === today.getFullYear()) {

                    dayDiv.classList.add('today');

                }


                // Selection Logic (if we were in a different selected state, we'd store state)

                // For simplicity, this calendar resets selection to "Today" every time we open

                // To make it persist, we would add an event listener here.

                // For this widget, clicking any date sets that date as "Selected".

                

                dayDiv.onclick = () => {

                    // Update details text

                    // We need to calculate the string for the selected date dynamically

                    // For now, let's just update the container

                    const currentMonthName = monthNames[currentMonth];

                    detailMonth.textContent = currentMonthName;

                    detailFull.textContent = i;

                    

                    // Highlight logic

                    daysGrid.querySelectorAll('.day-cell').forEach(c => {

                        c.classList.remove('selected');

                    });

                    dayDiv.classList.add('selected');

                    

                    // Re-position "Today" dot based on current system date vs selected date

                    // (Note: For a pure demo without a "state" variable, we highlight Today AND the Clicked Day)

                    

                    // Visual Feedback for Animation

                    daysGrid.style.transform = 'scale(0.95)';

                    setTimeout(() => { daysGrid.style.transform = 'scale(1)'; }, 100);

                };


                gridContainer.appendChild(dayDiv);

            }

        }


        prevBtn.addEventListener('click', () => {

            currentDate.setMonth(currentDate.getMonth() - 1);

            renderCalendar();

        });


        nextBtn.addEventListener('click', () => {

            currentDate.setMonth(currentDate.getMonth() + 1);

            renderCalendar();

        });


        // Initialize

        renderCalendar();


        // --- SELECTED DATE LOGIC UPDATE ---

        // This simple script adds the ability to click a day, update the bottom text, and highlight it

        daysGrid.addEventListener('click', (e) => {

            // If clicking a specific day cell

            if(e.target.classList.contains('day-cell') && !e.target.classList.contains('empty')){

                e.target.classList.add('selected');

                // Update bottom text

                detailMonth.textContent = monthNames[currentDate.getMonth()];

                detailFull.textContent = e.target.textContent;

            }

        });


        // Add a "Reset to Today" button functionality via the "Today" class click?

        // Actually, let's listen for clicks on "Today" specifically to reset the selection

        document.addEventListener('click', (e) => {

            const todayBtn = document.querySelector('.day-cell.today');

            if(e.target === todayBtn && todayBtn.classList.contains('today')){

                // Logic to reset calendar date

                currentDate = new Date();

                renderCalendar();

            }

        });
